  # flama:begin extension
  test-extension:
    name: Extension in Chromium
    runs-on: gha-vm
    needs: check
    # The built extension, loaded into a real Chromium: Chrome accepts the
    # manifest, the service worker starts, the pages render and reach the
    # worker. The API is answered by the specs, so nothing else runs here.
    # The unit suites already ran in Check.
    if: contains(needs.check.outputs.packages, '"@flama/extension"')
    steps:
      - uses: actions/checkout@3d3c42e5aac5ba805825da76410c181273ba90b1 # v7.0.1
      - uses: pnpm/action-setup@ea17c68df8912ef543352723c149a84f56e3d413 # v6.1.0
      - uses: actions/setup-node@820762786026740c76f36085b0efc47a31fe5020 # v7.0.0
        with:
          node-version: 26
          cache: pnpm
      - run: pnpm install --frozen-lockfile
      - run: pnpm turbo run build --filter=@flama/extension
      - run: pnpm --filter @flama/extension exec playwright install --with-deps chromium
      - run: pnpm --filter @flama/extension e2e
      # The archive the Web Store takes, built and checked (no `key`, no source
      # maps, not a development build) on every change, so a release is never
      # the first time it is made.
      - run: pnpm --filter @flama/extension zip
  # flama:end extension
