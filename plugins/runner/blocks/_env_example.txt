# The one credential that exists before any key is issued. Holds every scope.

# Optional Postgres DSN. Empty keeps the in-memory stores (the default, fine

RUNNER_DATABASE_URL=
# Optional: enables short-lived HS256 service tokens (POST /v1/service-tokens)

# `iss`/`aud` claims required on service tokens (default flama-runner)
RUNNER_JWT_ISSUER=

# Service token lifetime (default 1h)
RUNNER_JWT_TTL=
# Reverse-proxy hops in front of the runner, same meaning as TRUST_PROXY

# Graceful shutdown budget for draining requests and WebSockets (default 15s)
RUNNER_SHUTDOWN_TIMEOUT=
# Maximum JSON request body in bytes (default 1048576)
