  # flama:begin runner
  runner:
    image: ${RUNNER_IMAGE:-ghcr.io/your-org/flama-runner:latest}
    restart: always
    depends_on:
      postgres:
        condition: service_healthy
    environment:
      RUNNER_ENV: production
      RUNNER_PORT: 3006
      # Empty keeps the in-memory stores; set it to persist keys and jobs.
      # A separate database on the same server keeps the runner's tables out
      # of the API's schema.
      RUNNER_DATABASE_URL: ${RUNNER_DATABASE_URL:-}
      # Required. The API authenticates to the runner with this key or one
      # minted from it; nothing else in the stack needs to know it.
      RUNNER_BOOTSTRAP_API_KEY: ${RUNNER_BOOTSTRAP_API_KEY}
      # Empty keeps service tokens off (see GET /health/capabilities).
      RUNNER_JWT_SECRET: ${RUNNER_JWT_SECRET:-}
      ERROR_TYPE_BASE_URL: ${ERROR_TYPE_BASE_URL:-https://flama.dev/errors}
    healthcheck:
      # The image is distroless (no shell, no curl), so the binary probes
      # itself: `runner healthcheck` GETs /healthz on RUNNER_PORT. Liveness,
      # not readiness: a full job queue fails /readyz, not this.
      test: ["CMD", "/runner", "healthcheck"]
      interval: 10s
      timeout: 5s
      retries: 5
      start_period: 10s
    # Not published: only the API (on the compose network) talks to it.
    expose:
      - "3006"
  # flama:end runner
