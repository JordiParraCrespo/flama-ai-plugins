# flama:begin runner
runner:
  image: ghcr.io/your-org/flama-runner:latest
  port: 3006
  resources:
    requests:
      cpu: 50m
      memory: 64Mi
    limits:
      cpu: 500m
      memory: 256Mi
  # RUNNER_BOOTSTRAP_API_KEY, RUNNER_JWT_SECRET and (optionally)
  # RUNNER_DATABASE_URL come from the <release>-runner-secrets Secret, never
  # from values. Without RUNNER_DATABASE_URL the runner stays stateless.
  env:
    RUNNER_ENV: production
    ERROR_TYPE_BASE_URL: https://docs.flama.dev/errors
# flama:end runner
