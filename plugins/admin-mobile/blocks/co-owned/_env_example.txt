# ------------------------------------------------------------------------------
# Mobile apps — EXPO_PUBLIC_* values are public and baked in at build time
# (each app's app.config.ts loads this file before Metro bundles)
# ------------------------------------------------------------------------------

# Use http://localhost:3001 for the iOS simulator. On a physical device or
# Android emulator, point this at your machine's LAN IP (e.g. http://192.168.x.x:3001)
# and add that origin to the API's trusted origins if needed.
EXPO_PUBLIC_API_URL=http://localhost:3001

# Analytics (PostHog) — same semantics as the VITE_ pair above
EXPO_PUBLIC_POSTHOG_KEY=
EXPO_PUBLIC_POSTHOG_HOST=

# Error reporting (Sentry). Empty disables the SDK. The native plugin still
# runs at prebuild; a missing DSN means nothing is sent.
EXPO_PUBLIC_SENTRY_DSN=
SENTRY_ORG=
SENTRY_PROJECT=

# RevenueCat. Empty disables IAP. iOS and Android use different keys.
EXPO_PUBLIC_REVENUECAT_IOS_KEY=
EXPO_PUBLIC_REVENUECAT_ANDROID_KEY=

# Optional remote config document (JSON). Static defaults always apply; a
# missing URL skips the remote layer.
EXPO_PUBLIC_CONFIG_URL=
